# archestra/agent_templates/catalog.py
"""
Quickstart catalog of pre-built agent templates.
Each template includes system prompt, model config, tool assignments,
and MCP server definitions for one-click agent setup.
"""
from __future__ import annotations

import json
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Any
from enum import Enum

from archestra.agents.base import AgentConfig
from archestra.tools.registry import ToolAssignment
from archestra.mcp.server import MCPServerConfig


class TemplateCategory(str, Enum):
    """Categories for organizing agent templates."""
    CUSTOMER_SUPPORT = "customer_support"
    CODE_ASSISTANT = "code_assistant"
    DATA_ANALYST = "data_analyst"
    RESEARCH = "research"
    CREATIVE = "creative"
    PRODUCTIVITY = "productivity"


@dataclass
class AgentTemplate:
    """
    A pre-built agent template with all configuration needed to spin up an agent.
    
    Attributes:
        id: Unique identifier for the template.
        name: Human-readable name.
        description: Short description of the template's purpose.
        category: Category for filtering.
        system_prompt: The system prompt for the agent.
        model: Model identifier (e.g., "gpt-4", "claude-3-opus").
        temperature: Sampling temperature.
        max_tokens: Maximum tokens per response.
        tool_assignments: List of tool assignments (tool name + config).
        mcp_servers: List of MCP server configurations to install.
        tags: Optional tags for search/filter.
        version: Template version for upgrade tracking.
    """
    id: str
    name: str
    description: str
    category: TemplateCategory
    system_prompt: str
    model: str = "gpt-4"
    temperature: float = 0.7
    max_tokens: int = 2048
    tool_assignments: List[ToolAssignment] = field(default_factory=list)
    mcp_servers: List[MCPServerConfig] = field(default_factory=list)
    tags: List[str] = field(default_factory=list)
    version: str = "1.0.0"

    def to_agent_config(self) -> AgentConfig:
        """Convert template to an AgentConfig for agent instantiation."""
        return AgentConfig(
            system_prompt=self.system_prompt,
            model=self.model,
            temperature=self.temperature,
            max_tokens=self.max_tokens,
            tool_assignments=self.tool_assignments,
            mcp_servers=self.mcp_servers,
        )

    def to_dict(self) -> Dict[str, Any]:
        """Serialize to dictionary (excluding internal fields)."""
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "AgentTemplate":
        """Deserialize from dictionary."""
        return cls(**data)
