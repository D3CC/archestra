  /**
   * Register a new agent template
   */
  registerTemplate(template: AgentTemplate): void {
    const validationResult = AgentTemplateSchema.safeParse(template);
    if (!validationResult.success) {
      logger.error('Invalid agent template:', validationResult.error);
      throw new Error(`Invalid template: ${validationResult.error.message}`);
    }

    if (this.templates.has(template.id)) {
      logger.warn(`Template ${template.id} already exists, overwriting`);
    }

    this.templates.set(template.id, template);
    logger.info(`Registered template: ${template.id} (${template.name})`);
  }

  /**
   * Get a template by ID
   */
  getTemplate(id: string): AgentTemplate | undefined {
    return this.templates.get(id);
  }

  /**
   * List all available templates
   */
  listTemplates(category?: string): AgentTemplate[] {
    const allTemplates = Array.from(this.templates.values());
    if (category) {
      return allTemplates.filter(t => t.category === category);
    }
    return allTemplates;
  }

  /**
   * Search templates by keyword
   */
  searchTemplates(query: string): AgentTemplate[] {
    const lowerQuery = query.toLowerCase();
    return Array.from(this.templates.values()).filter(template => {
      return (
        template.name.toLowerCase().includes(lowerQuery) ||
        template.description.toLowerCase().includes(lowerQuery) ||
        template.tags.some(tag => tag.toLowerCase().includes(lowerQuery))
      );
    });
  }

  /**
   * Create an agent configuration from a template
   */
  createAgentFromTemplate(templateId: string, overrides?: Partial<AgentConfig>): AgentConfig {
    const template = this.getTemplate(templateId);
    if (!template) {
      throw new Error(`Template ${templateId} not found`);
    }

    const agentConfig: AgentConfig = {
      systemPrompt: template.systemPrompt,
      model: template.model,
      tools: [...template.tools],
      mcpServers: template.mcpServers.map(server => ({
        name: server.name,
        url: server.url,
        config: {},
      })),
      ...overrides,
    };

    logger.info(`Created agent config from template: ${templateId}`);
    return agentConfig;
  }
