/**
 * Catalog of agent templates for quickstart deployment.
 */
export class AgentTemplateCatalog {
  private templates: Map<string, AgentTemplate> = new Map();
  private mcpClient: MCPClient;

  constructor(mcpClient: MCPClient) {
    this.mcpClient = mcpClient;
    this.loadBuiltinTemplates();
  }

  private loadBuiltinTemplates(): void {
    for (const template of BUILTIN_TEMPLATES) {
      const parsed = AgentTemplateSchema.safeParse(template);
      if (parsed.success) {
        this.templates.set(parsed.data.id, parsed.data);
      } else {
        logger.warn(`Failed to parse builtin template ${template.id}: ${parsed.error.message}`);
      }
    }
  }

  /**
   * Returns all available templates, optionally filtered by category.
   */
  listTemplates(category?: string): AgentTemplate[] {
    const all = Array.from(this.templates.values());
    if (category) {
      return all.filter(t => t.category === category);
    }
    return all;
  }

  /**
   * Retrieves a template by its unique ID.
   */
  getTemplate(id: string): AgentTemplate | undefined {
    return this.templates.get(id);
  }

  /**
   * Registers a custom template at runtime.
   */
  registerTemplate(template: AgentTemplate): void {
    const parsed = AgentTemplateSchema.safeParse(template);
    if (!parsed.success) {
      throw new Error(`Invalid template: ${parsed.error.message}`);
    }
    if (this.templates.has(parsed.data.id)) {
      logger.warn(`Overwriting existing template: ${parsed.data.id}`);
    }
    this.templates.set(parsed.data.id, parsed.data);
    logger.info(`Registered template: ${parsed.data.id}`);
  }

  /**
   * Deploys an agent from a template: installs MCP servers and returns the agent configuration.
   */
  async deployTemplate(id: string): Promise<{ agentConfig: Record<string, unknown>; mcpStatus: string[] }> {
    const template = this.templates.get(id);
    if (!template) {
      throw new Error(`Template not found: ${id}`);
    }

    const mcpStatus: string[] = [];
    for (const server of template.mcpServers) {
      try {
        await this.mcpClient.installServer(server);
        mcpStatus.push(`Installed ${server.name}`);
      } catch (error) {
        const msg = `Failed to install ${server.name}: ${(error as Error).message}`;
        logger.error(msg);
        mcpStatus.push(msg);
      }
    }

    const agentConfig = {
      systemPrompt: template.systemPrompt,
      model: template.model,
      tools: template.tools.map(t => ({
        name: t.name,
        description: t.description,
        parameters: t.parameters,
      })),
      templateId: template.id,
    };

    return { agentConfig, mcpStatus };
  }
