"""
Agent Template Catalog - Quickstart catalog of pre-built agent templates.

This module provides a catalog of pre-configured agent templates that users
can deploy with a single click. Each template includes system prompt, model
configuration, tool assignments, and MCP server installation instructions.
"""

# Data Models

class AgentCategory(str, Enum):
    """Categories for agent templates."""
    GENERAL = "general"
    CODING = "coding"
    DATA_SCIENCE = "data_science"
    DEVOPS = "devops"
    WRITING = "writing"
    CUSTOMER_SUPPORT = "customer_support"
    RESEARCH = "research"


@dataclass
class MCPServerTemplate:
    """Template for an MCP server that the agent requires."""
    name: str
    description: str
    install_command: str
    config: Dict[str, Any] = field(default_factory=dict)
    version: str = "latest"

    def to_mcp_config(self) -> MCPServerConfig:
        return MCPServerConfig(
            name=self.name,
            command=self.install_command.split()[0],
            args=self.install_command.split()[1:],
            env=self.config.get("env", {}),
        )


@dataclass
class AgentTemplate:
    """A pre-built agent template for quickstart deployment."""
    id: str
    name: str
    description: str
    category: AgentCategory
    system_prompt: str
    model: str
    temperature: float = 0.7
    max_tokens: int = 4096
    tools: List[str] = field(default_factory=list)
    mcp_servers: List[MCPServerTemplate] = field(default_factory=list)
    tags: List[str] = field(default_factory=list)
    version: str = "1.0.0"
    author: str = "archestra"

    def to_agent_config(self, tool_registry: Optional[ToolRegistry] = None) -> AgentConfig:
        """Convert this template to a full AgentConfig."""
        config = AgentConfig(
            name=self.name,
            system_prompt=self.system_prompt,
            model=self.model,
            temperature=self.temperature,
            max_tokens=self.max_tokens,
            tools=[],
            mcp_servers=[s.to_mcp_config() for s in self.mcp_servers],
        )
        if tool_registry:
            for tool_name in self.tools:
                tool = tool_registry.get(tool_name)
                if tool:
                    config.tools.append(tool)
                else:
                    logger.warning(f"Tool '{tool_name}' not found in registry, skipping.")
        return config

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "AgentTemplate":
        mcp_servers = [MCPServerTemplate(**s) for s in data.pop("mcp_servers", [])]
        data["category"] = AgentCategory(data["category"])
        return cls(mcp_servers=mcp_servers, **data)

class TemplateCatalog:
    """Catalog of agent templates with load/save capabilities."""

    def __init__(self, templates: Optional[List[AgentTemplate]] = None):
        self._templates: Dict[str, AgentTemplate] = {}
        if templates:
            for t in templates:
                self._templates[t.id] = t

    @classmethod
    def with_builtins(cls) -> "TemplateCatalog":
        """Create a catalog pre-loaded with built-in templates."""
        return cls(BUILTIN_TEMPLATES)
