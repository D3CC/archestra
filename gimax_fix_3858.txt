"""
Agent template catalog module for quickstart deployment of pre-built agents.
Provides a registry of agent templates with system prompts, model configs,
tool assignments, and MCP server installation support.
"""

from archestra.mcp.server import MCPServerConfig, install_mcp_server

logger = logging.getLogger(__name__)

# --------------------------------------------------------------------------- #
# Data Models
# --------------------------------------------------------------------------- #

@dataclass
class AgentTemplate:
    """Represents a pre-built agent template for quickstart deployment."""
    id: str
    name: str
    description: str
    system_prompt: str
    model: str
    tools: List[ToolAssignment] = field(default_factory=list)
    mcp_servers: List[MCPServerConfig] = field(default_factory=list)
    tags: List[str] = field(default_factory=list)
    version: str = "1.0.0"
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_agent_config(self) -> AgentConfig:
        """Convert template to an AgentConfig for instantiation."""
        return AgentConfig(
            name=self.name,
            system_prompt=self.system_prompt,
            model=self.model,
            tools=self.tools,
        )

    def to_dict(self) -> Dict[str, Any]:
        """Serialize template to dictionary."""
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "AgentTemplate":
        """Deserialize dictionary to AgentTemplate."""
        tools = [ToolAssignment(**t) if isinstance(t, dict) else t for t in data.get("tools", [])]
        mcp_servers = [
            MCPServerConfig(**s) if isinstance(s, dict) else s
            for s in data.get("mcp_servers", [])
        ]
        return cls(
            id=data["id"],
            name=data["name"],
            description=data.get("description", ""),
            system_prompt=data["system_prompt"],
            model=data["model"],
            tools=tools,
            mcp_servers=mcp_servers,
            tags=data.get("tags", []),
            version=data.get("version", "1.0.0"),
            metadata=data.get("metadata", {}),
        )


# --------------------------------------------------------------------------- #
# Catalog Registry
# --------------------------------------------------------------------------- #

class AgentTemplateCatalog:
    """
    Registry of agent templates. Supports loading from YAML/JSON files,
    searching by tags, and deploying templates with MCP server installation.
    """

    def __init__(self, templates: Optional[List[AgentTemplate]] = None):
        self._templates: Dict[str, AgentTemplate] = {}
        if templates:
            for tmpl in templates:
                self.register(tmpl)

    def register(self, template: AgentTemplate) -> None:
        """Register a template in the catalog."""
        if template.id in self._templates:
            logger.warning("Overwriting existing template: %s", template.id)
        self._templates[template.id] = template

    def get(self, template_id: str) -> Optional[AgentTemplate]:
        """Retrieve a template by ID."""
        return self._templates.get(template_id)

    def list_templates(self) -> List[AgentTemplate]:
        """Return all registered templates."""
        return list(self._templates.values())

    def search_by_tag(self, tag: str) -> List[AgentTemplate]:
        """Find templates that contain a specific tag."""
        return [t for t in self._templates.values() if tag in t.tags]

    def load_from_file(self, path: str) -> None:
        """Load templates from a YAML or JSON file."""
        p = Path(path)
        if not p.exists():
            raise FileNotFoundError(f"Template file not found: {path}")

        with open(p, "r") as f:
            if p.suffix in (".yaml", ".yml"):
                data = yaml.safe_load(f)
            elif p.suffix == ".json":
                data = json.load(f)
            else:
                raise ValueError(f"Unsupported file format: {p.suffix}")

        templates_data = data if isinstance(data, list) else data.get("templates", [])
        for item in templates_data:
            template = AgentTemplate.from_dict(item)
            self.register(template)

    def deploy_template(
        self,
        template_id: str,
        install_mcp: bool = True,
        **kwargs: Any,
    ) -> AgentConfig:
        """
        Deploy a template: return an AgentConfig and optionally install MCP servers.
        Extra kwargs override template fields (e.g., model, system_prompt).
        """
        template = self.get(template_id)
        if template is None:
            raise ValueError(f"Template not found: {template_id}")

        # Apply overrides
        overridden = template.to_dict()
        overridden.update(kwargs)
        final_template = AgentTemplate.from_dict(overridden)

        if install_mcp:
            for mcp_config in final_template.mcp_servers:
                logger.info("Installing MCP server: %s", mcp_config.name)
                install_mcp_server(mcp_config)

        return final_template.to_agent_config()

# --------------------------------------------------------------------------- #
# Convenience Functions
# --------------------------------------------------------------------------- #

def quickstart_deploy(template_id: str, **overrides: Any) -> AgentConfig:
    """One-liner to deploy a built-in template with optional overrides."""
    catalog = get_default_catalog()
    return catalog.deploy_template(template_id, **overrides)

def test_catalog_operations():
    """Test basic catalog operations."""
    catalog = get_default_catalog()
    assert len(catalog.list_templates()) == 3

    tmpl = catalog.get("code-assistant")
    assert tmpl is not None
    assert "coding" in tmpl.tags

    results = catalog.search_by_tag("quickstart")
    assert len(results) == 3

    results = catalog.search_by_tag("data")
    assert len(results) == 1
    assert results[0].id == "data-analyst"
