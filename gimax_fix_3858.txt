// src/catalog/agent-template-catalog.ts
import { z } from 'zod';
import { AgentConfig, AgentRole, ModelProvider, ToolAssignment } from '../agent/types';
import { MCPServerConfig } from '../mcp/types';

/**
 * Schema for an agent template in the catalog.
 */
export const AgentTemplateSchema = z.object({
  id: z.string().uuid(),
  name: z.string().min(1).max(100),
  description: z.string().max(500).optional(),
  category: z.enum(['general', 'coding', 'writing', 'data', 'devops', 'customer-support']),
  systemPrompt: z.string().min(1),
  model: z.object({
    provider: z.nativeEnum(ModelProvider),
    modelId: z.string(),
    temperature: z.number().min(0).max(2).optional(),
    maxTokens: z.number().int().positive().optional(),
  }),
  tools: z.array(z.object({
    toolId: z.string(),
    enabled: z.boolean().default(true),
    config: z.record(z.unknown()).optional(),
  })),
  mcpServers: z.array(z.object({
    serverId: z.string(),
    url: z.string().url(),
    config: z.record(z.unknown()).optional(),
  })),
  metadata: z.object({
    author: z.string().optional(),
    version: z.string().regex(/^\d+\.\d+\.\d+$/),
    createdAt: z.string().datetime(),
    updatedAt: z.string().datetime(),
    tags: z.array(z.string()).optional(),
  }),
});

export type AgentTemplate = z.infer<typeof AgentTemplateSchema>;

/**
 * In-memory catalog of pre-built agent templates.
 * In production, this would be backed by a database or remote registry.
 */
export class AgentTemplateCatalog {
  private templates: Map<string, AgentTemplate> = new Map();

  constructor(initialTemplates?: AgentTemplate[]) {
    if (initialTemplates) {
      for (const tmpl of initialTemplates) {
        this.addTemplate(tmpl);
      }
    }
  }

  /**
   * Add a template to the catalog. Validates against schema.
   */
  addTemplate(template: AgentTemplate): void {
    const parsed = AgentTemplateSchema.parse(template);
    this.templates.set(parsed.id, parsed);
  }

  /**
   * Retrieve a template by ID.
   */
  getTemplate(id: string): AgentTemplate | undefined {
    return this.templates.get(id);
  }

  /**
   * List all templates, optionally filtered by category.
   */
  listTemplates(category?: string): AgentTemplate[] {
    const all = Array.from(this.templates.values());
    if (category) {
      return all.filter(t => t.category === category);
    }
    return all;
  }

  /**
   * Remove a template from the catalog.
   */
  removeTemplate(id: string): boolean {
    return this.templates.delete(id);
  }

  /**
   * Convert a template into a fully configured AgentConfig and MCPServerConfig list.
   * This is the "single click" instantiation.
   */
  instantiate(templateId: string): { agentConfig: AgentConfig; mcpServers: MCPServerConfig[] } | null {
    const template = this.getTemplate(templateId);
    if (!template) return null;

    const agentConfig: AgentConfig = {
      role: AgentRole.ASSISTANT,
      systemPrompt: template.systemPrompt,
      model: {
        provider: template.model.provider,
        modelId: template.model.modelId,
        temperature: template.model.temperature ?? 0.7,
        maxTokens: template.model.maxTokens ?? 4096,
      },
      tools: template.tools.map(t => ({
        toolId: t.toolId,
        enabled: t.enabled,
        config: t.config ?? {},
      })) as ToolAssignment[],
    };

    const mcpServers: MCPServerConfig[] = template.mcpServers.map(s => ({
      serverId: s.serverId,
      url: s.url,
      config: s.config ?? {},
    }));

    return { agentConfig, mcpServers };
  }
}
