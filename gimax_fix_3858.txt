"""
Agent Template Catalog - Quickstart catalog of pre-built agent templates.

This module provides a catalog of pre-configured agent templates that users can
deploy with a single click. Each template includes system prompt, model configuration,
tool assignments, and MCP server installations.

Usage:
    from archestra.agents.templates import TemplateCatalog, AgentTemplate
    
    catalog = TemplateCatalog()
    templates = catalog.list_templates()
    agent = catalog.instantiate("research-assistant")
"""

from __future__ import annotations

import json
import logging
from dataclasses import dataclass, field, asdict
from enum import Enum
from pathlib import Path
from typing import Any, Dict, List, Optional

logger = logging.getLogger(__name__)


class TemplateCategory(str, Enum):
    """Categories for agent templates."""
    RESEARCH = "research"
    CODING = "coding"
    WRITING = "writing"
    DATA_ANALYSIS = "data_analysis"
    CUSTOMER_SUPPORT = "customer_support"
    PRODUCTIVITY = "productivity"
    CUSTOM = "custom"


@dataclass
class MCPServerConfig:
    """Configuration for an MCP server to install with the agent."""
    name: str
    command: str
    args: List[str] = field(default_factory=list)
    env_vars: Dict[str, str] = field(default_factory=dict)
    description: str = ""


@dataclass
class AgentTemplate:
    """
    A pre-built agent template with all configuration needed to spin up an agent.
    
    Attributes:
        id: Unique identifier for the template.
        name: Human-readable name.
        description: Brief description of what this agent does.
        category: Category of the template.
        system_prompt: The system prompt for the agent.
        model: Model identifier (e.g., "gpt-4", "claude-3-opus").
        tools: List of tool names to assign.
        mcp_servers: List of MCP server configurations to install.
        parameters: Additional model parameters (temperature, max_tokens, etc.).
        tags: List of tags for filtering/searching.
        version: Template version string.
        author: Template author.
    """
    id: str
    name: str
    description: str
    category: TemplateCategory = TemplateCategory.CUSTOM
    system_prompt: str = ""
    model: str = "gpt-4"
    tools: List[str] = field(default_factory=list)
    mcp_servers: List[MCPServerConfig] = field(default_factory=list)
    parameters: Dict[str, Any] = field(default_factory=lambda: {
        "temperature": 0.7,
        "max_tokens": 4096,
    })
    tags: List[str] = field(default_factory=list)
    version: str = "1.0.0"
    author: str = "Archestra AI"

    def to_dict(self) -> Dict[str, Any]:
        """Serialize template to dictionary."""
        result = asdict(self)
        result["category"] = self.category.value
        result["mcp_servers"] = [
            {**asdict(server), "env_vars": dict(server.env_vars)}
            for server in self.mcp_servers
        ]
        return result

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "AgentTemplate":
        """Deserialize template from dictionary."""
        data = data.copy()
        data["category"] = TemplateCategory(data.get("category", "custom"))
        mcp_servers_data = data.pop("mcp_servers", [])
        data["mcp_servers"] = [MCPServerConfig(**s) for s in mcp_servers_data]
        return cls(**data)

        return list(self._templates.values())
