"""
Agent template catalog for quick-start agent creation.
Provides pre-built agent templates with system prompts, model configs,
tool assignments, and MCP server installation in a single click.
"""

# Data models

@dataclass
class MCPServerSpec:
    """Specification for an MCP server to install with the agent."""
    name: str
    command: str
    args: List[str] = field(default_factory=list)
    env: Dict[str, str] = field(default_factory=dict)
    description: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "MCPServerSpec":
        return cls(**data)


@dataclass
class AgentTemplate:
    """A pre-built agent template for quick deployment."""
    id: str
    name: str
    description: str
    system_prompt: str
    model: str
    temperature: float = 0.7
    max_tokens: int = 2048
    tools: List[str] = field(default_factory=list)
    mcp_servers: List[MCPServerSpec] = field(default_factory=list)
    tags: List[str] = field(default_factory=list)
    version: str = "1.0.0"

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "description": self.description,
            "system_prompt": self.system_prompt,
            "model": self.model,
            "temperature": self.temperature,
            "max_tokens": self.max_tokens,
            "tools": self.tools,
            "mcp_servers": [s.to_dict() for s in self.mcp_servers],
            "tags": self.tags,
            "version": self.version,
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "AgentTemplate":
        mcp_servers = [MCPServerSpec.from_dict(s) for s in data.pop("mcp_servers", [])]
        return cls(mcp_servers=mcp_servers, **data)

    def create_agent_config(self) -> AgentConfig:
        """Convert template to an AgentConfig for agent instantiation."""
        return AgentConfig(
            system_prompt=self.system_prompt,
            model=self.model,
            temperature=self.temperature,
            max_tokens=self.max_tokens,
            tools=self.tools,
        )

    # ------------------------------------------------------------------
    # Load / save
    # ------------------------------------------------------------------

    @classmethod
    def load_from_file(cls, path: Path) -> "TemplateCatalog":
        """Load templates from a YAML or JSON file."""
        if not path.exists():
            raise FileNotFoundError(f"Template file not found: {path}")

        with open(path, "r") as f:
            if path.suffix in (".yaml", ".yml"):
                data = yaml.safe_load(f)
            elif path.suffix == ".json":
                data = json.load(f)
            else:
                raise ValueError(f"Unsupported file format: {path.suffix}")

        templates = {}
        for item in data.get("templates", []):
            template = AgentTemplate.from_dict(item)
            templates[template.id] = template
        return cls(templates=templates)

    def save_to_file(self, path: Path) -> None:
        """Save catalog to a YAML file."""
        data = {
            "version": "1.0",
            "templates": [t.to_dict() for t in self._templates.values()],
        }
        with open(path, "w") as f:
            yaml.dump(data, f, default_flow_style=False, sort_keys=False)

    # ------------------------------------------------------------------
    # Query
    # ------------------------------------------------------------------

    def get(self, template_id: str) -> Optional[AgentTemplate]:
        return self._templates.get(template_id)

    def list_templates(self, tag: Optional[str] = None) -> List[AgentTemplate]:
        if tag:
            return [t for t in self._templates.values() if tag in t.tags]
        return list(self._templates.values())

    def add_template(self, template: AgentTemplate) -> None:
        if template.id in self._templates:
            logger.warning("Overwriting existing template: %s", template.id)
        self._templates[template.id] = template

    def remove_template(self, template_id: str) -> bool:
        return self._templates.pop(template_id, None) is not None
