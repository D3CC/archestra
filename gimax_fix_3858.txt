export function getQuickstartCatalog(): AgentTemplate[] {
  return [...QUICKSTART_CATALOG];
}

export function getTemplateById(id: string): AgentTemplate | undefined {
  return QUICKSTART_CATALOG.find(template => template.id === id);
}

export function getTemplatesByCategory(category: string): AgentTemplate[] {
  return QUICKSTART_CATALOG.filter(template => template.category === category);
}

export function searchTemplates(query: string): AgentTemplate[] {
  const lowerQuery = query.toLowerCase();
  return QUICKSTART_CATALOG.filter(template =>
    template.name.toLowerCase().includes(lowerQuery) ||
    template.description.toLowerCase().includes(lowerQuery) ||
    template.tags.some(tag => tag.toLowerCase().includes(lowerQuery))
  );
}

import { AgentTemplate, MCPServerConfig } from './types';

export interface InstallationResult {
  success: boolean;
  agentId?: string;
  mcpResults: Array<{
    serverName: string;
    installed: boolean;
    error?: string;
  }>;
  error?: string;
}

export class AgentInstaller {
  private installedAgents: Map<string, boolean> = new Map();
  private installedMcpServers: Map<string, boolean> = new Map();

  async installAgent(template: AgentTemplate): Promise<InstallationResult> {
    const result: InstallationResult = {
      success: false,
      mcpResults: []
    };

    try {
      // Validate template
      this.validateTemplate(template);

      // Install MCP servers first
      for (const server of template.mcpServers) {
        try {
          await this.installMCPServer(server);
          result.mcpResults.push({
            serverName: server.name,
            installed: true
          });
          this.installedMcpServers.set(server.name, true);
        } catch (error) {
          result.mcpResults.push({
            serverName: server.name,
            installed: false,
            error: error instanceof Error ? error.message : 'Unknown error'
          });
        }
      }

      // Configure agent
      const agentId = await this.configureAgent(template);
      this.installedAgents.set(agentId, true);

      result.success = true;
      result.agentId = agentId;
    } catch (error) {
      result.error = error instanceof Error ? error.message : 'Installation failed';
    }

    return result;
  }

  private validateTemplate(template: AgentTemplate): void {
    if (!template.id || !template.name || !template.systemPrompt || !template.model) {
      throw new Error('Invalid template: missing required fields');
    }

    if
