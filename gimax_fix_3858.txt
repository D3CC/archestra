  async getTemplate(id: string): Promise<AgentTemplate | null> {
    const template = this.templates.get(id);
    if (!template) {
      logger.warn(`Template not found: ${id}`);
      return null;
    }
    return template;
  }

  async listTemplates(category?: string): Promise<AgentTemplate[]> {
    const allTemplates = Array.from(this.templates.values());
    if (category) {
      return allTemplates.filter(t => t.category === category);
    }
    return allTemplates;
  }

  async searchTemplates(query: string): Promise<AgentTemplate[]> {
    const lowerQuery = query.toLowerCase();
    return Array.from(this.templates.values()).filter(template => {
      return (
        template.name.toLowerCase().includes(lowerQuery) ||
        template.description.toLowerCase().includes(lowerQuery) ||
        template.tags?.some(tag => tag.toLowerCase().includes(lowerQuery))
      );
    });
  }

  async createAgentFromTemplate(
    templateId: string,
    overrides?: Partial<AgentConfig>
  ): Promise<AgentConfig> {
    const template = await this.getTemplate(templateId);
    if (!template) {
      throw new Error(`Template not found: ${templateId}`);
    }

    const agentConfig: AgentConfig = {
      name: overrides?.name || template.name,
      systemPrompt: overrides?.systemPrompt || template.systemPrompt,
      model: overrides?.model || template.model,
      tools: overrides?.tools || template.tools,
      mcpServers: overrides?.mcpServers || template.mcpServers,
      metadata: {
        templateId: template.id,
        templateVersion: template.version,
        createdAt: new Date().toISOString(),
      },
    };

    logger.info(`Agent created from template: ${template.name} (${templateId})`);
    return agentConfig;
  }

  async installMCPServers(templateId: string): Promise<boolean> {
    const template = await this.getTemplate(templateId);
    if (!template) {
      throw new Error(`Template not found: ${templateId}`);
    }

    try {
      for (const server of template.mcpServers) {
        logger.info(`Installing MCP server: ${server.name} at ${server.url}`);
        // Simulate MCP server installation
        await this.installMCPServer(server);
      }
      logger.info(`All MCP servers installed for template: ${template.name}`);
      return true;
    } catch (error) {
      logger.error(`Failed to install MCP servers for template: ${template.name}`, error);
      throw error;
    }
  }

  private async installMCPServer(server: MCPConfig): Promise<void> {
    // Simulate async installation process
    return new Promise((resolve) => {
      setTimeout(() => {
        logger.debug(`MCP server ${server.name} installed successfully`);
        resolve();
      }, 100);
    });
  }
