   * Get all available templates
   */
  getAllTemplates(): AgentTemplateRecord[] {
    return Array.from(this.templates.values());
  }

  /**
   * Get templates by category
   */
  getTemplatesByCategory(category: string): AgentTemplateRecord[] {
    return this.getAllTemplates().filter(t => t.category === category);
  }

  /**
   * Get a specific template by ID
   */
  getTemplateById(id: string): AgentTemplateRecord | undefined {
    return this.templates.get(id);
  }

  /**
   * Add a custom template to the catalog
   */
  addTemplate(template: Omit<AgentTemplateRecord, 'id' | 'metadata.createdAt' | 'metadata.updatedAt'>): AgentTemplateRecord {
    const now = new Date().toISOString();
    const newTemplate: AgentTemplateRecord = {
      ...template,
      id: uuidv4(),
      metadata: {
        ...template.metadata,
        createdAt: now,
        updatedAt: now,
      },
    };

    const validationResult = AgentTemplateSchema.safeParse(newTemplate);
    if (!validationResult.success) {
      throw new Error(`Invalid template: ${validationResult.error.message}`);
    }

    this.templates.set(newTemplate.id, newTemplate);
    logger.info(`Added template: ${newTemplate.name} (${newTemplate.id})`);
    return newTemplate;
  }

  /**
   * Remove a template from the catalog
   */
  removeTemplate(id: string): boolean {
    const removed = this.templates.delete(id);
    if (removed) {
      logger.info(`Removed template: ${id}`);
    }
    return removed;
  }
