// Types for Windmill API interactions
interface WindmillWorkflow {
  id: string;
  name: string;
  description: string;
  nodes: WorkflowNode[];
  edges: WorkflowEdge[];
}

interface WorkflowNode {
  id: string;
  type: string;
  label: string;
  config: Record<string, unknown>;
  position: { x: number; y: number };
}

interface WorkflowEdge {
  id: string;
  source: string;
  target: string;
  label?: string;
}

interface MCPApp {
  id: string;
  name: string;
  description: string;
  workflowId: string;
  interactive: boolean;
  editable: boolean;
  nodes: WorkflowNode[];
  edges: WorkflowEdge[];
}

// Zod schemas for tool inputs
const CreateWorkflowSchema = z.object({
  name: z.string().min(1).max(100),
  description: z.string().optional(),
  nodes: z.array(z.object({
    type: z.enum(['confluence-get', 'email-send', 'http-request', 'transform', 'condition']),
    label: z.string(),
    config: z.record(z.unknown()).optional(),
  })).min(1).max(50),
});

const UpdateWorkflowSchema = z.object({
  workflowId: z.string().uuid(),
  nodes: z.array(z.object({
    id: z.string().uuid(),
    type: z.enum(['confluence-get', 'email-send', 'http-request', 'transform', 'condition']),
    label: z.string(),
    config: z.record(z.unknown()).optional(),
    position: z.object({
      x: z.number(),
      y: z.number(),
    }).optional(),
  })).optional(),
  edges: z.array(z.object({
    source: z.string().uuid(),
    target: z.string().uuid(),
    label: z.string().optional(),
  })).optional(),
});

const ExecuteWorkflowSchema = z.object({
  workflowId: z.string().uuid(),
  input: z.record(z.unknown()).optional(),

  appId: z.string().uuid(),

const workflows = new Map<string, WindmillWorkflow>();
const apps = new Map<string, MCPApp>();

// Helper to generate UUIDs
function generateId(): string {
  return crypto.randomUUID();
}

// Helper to create default nodes for common patterns
function createDefaultNodes(type: string): WorkflowNode[] {
  const nodes: WorkflowNode[] = [];
  const baseY = 100;
  const spacing = 150;

  switch (type) {
    case 'confluence-to-email':
      nodes.push({
        id: generateId(),
        type: 'confluence-get',
        label: 'Get Confluence Page',
        config: {
          pageId: '',
          space: '',
          title: '',
        },
        position: { x: 100, y: baseY },
      });
      nodes.push({
        id: generateId(),
        type: 'transform',
        label: 'Format Content',
        config: {
          template: '{{content}}',
        },
        position: { x: 100, y: baseY + spacing },
      });
      nodes.push({
        id: generateId(),
        type: 'email-send',
        label: 'Send Email',
        config: {
          to: '',
          subject: 'Confluence Update: {{title}}',
          body: '{{formattedContent}}',
        },
        position: { x: 100, y: baseY + 2 * spacing },
      });
      break;
    default:
      nodes.push({
        id: generateId(),
        type: 'http-request',
        label: 'HTTP Request',
        config: {
          url: '',
          method: 'GET',
          headers: {},
        },
        position: { x: 100, y: baseY },
      });
  }

  return nodes;
}

// Helper to create edges between sequential nodes
function createSequentialEdges(nodeIds: string[]): WorkflowEdge[] {
  const edges: WorkflowEdge[] = [];
  for (let i = 0; i < nodeIds.length - 1; i++) {
    edges.push({
      id: generateId(),
      source: nodeIds[i],
      target: nodeIds[i + 1],
    });
  }
  return edges;
}

        }));

          edges,

        const app: MCPApp = {
          id: appId,
          name: input.name,

            break;

              text: JSON.stringify({
                success: true,
