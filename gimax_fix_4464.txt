# archestra/models/mixins/soft_delete.py
"""
Soft delete mixin for all models.
Provides a generic way to mark records as deleted instead of hard-deleting them.
"""
from datetime import datetime, timezone
from typing import Optional
from sqlalchemy import DateTime, Boolean, event
from sqlalchemy.orm import declared_attr, Query
from sqlalchemy.ext.hybrid import hybrid_property
from archestra.extensions import db


class SoftDeleteMixin:
    """
    Mixin that adds soft delete capability to any model.
    
    Adds:
    - deleted_at column (datetime, nullable)
    - is_deleted property
    - soft_delete() method
    - restore() method
    - Automatic filtering of non-deleted records in queries
    """
    
    @declared_attr
    def deleted_at(cls):
        return db.Column(DateTime(timezone=True), nullable=True, default=None)
    
    @hybrid_property
    def is_deleted(self) -> bool:
        """Check if the record is soft-deleted."""
        return self.deleted_at is not None
    
    @is_deleted.expression
    def is_deleted(cls):
        """SQL expression for is_deleted."""
        return cls.deleted_at.isnot(None)
    
    def soft_delete(self) -> None:
        """Mark the record as deleted."""
        self.deleted_at = datetime.now(timezone.utc)
    
    def restore(self) -> None:
        """Restore a soft-deleted record."""
        self.deleted_at = None
    
    @classmethod
    def __init_subclass__(cls, **kwargs):
        """Automatically set up query filtering for subclasses."""
        super().__init_subclass__(**kwargs)
        # Add a query property that filters out deleted records by default
        if not hasattr(cls, 'query') or cls.query is None:
            cls.query = db.session.query_property()
        
        # Create a custom query class that filters soft-deleted records
        class SoftDeleteQuery(Query):
            def __new__(cls, *args, **kwargs):
                obj = super().__new__(cls)
                obj._soft_delete_filter = True
                return obj
            
            def __init__(self, *args, **kwargs):
                super().__init__(*args, **kwargs)
                self._soft_delete_filter = True
            
            def with_deleted(self):
                """Include soft-deleted records in the query."""
                self._soft_delete_filter = False
                return self
            
            def __iter__(self):
                """Override iteration to apply filter."""
                if self._soft_delete_filter and hasattr(self._entity_zero().class_, 'deleted_at'):
                    self = self.filter(
                        self._entity_zero().class_.deleted_at.is_(None)
                    )
                return super().__iter__()
        
        # Replace the query class
        cls.query_class = SoftDeleteQuery

# archestra/api/v1/resources.py (example API resource)
"""
API resource with soft delete handling.
"""
from flask_restful import Resource
from archestra.extensions import db
from archestra.models import User, Project


class UserResource(Resource):
    def get(self, user_id):
        """Get a user (soft-deleted users are excluded by default)."""
        user = User.query.get_or_404(user_id)
        return user.to_dict()
    
    def delete(self, user_id):
        """Soft delete a user."""
        user = User.query.get_or_404(user_id)
        user.soft_delete()
        db.session.commit()
        return {'message': 'User soft deleted'}, 200
    
    def post(self, user_id, action):
        """Restore a soft-deleted user."""
        if action == 'restore':
            user = User.query.with_deleted().get_or_404(user_id)
            if user.is_deleted:
                user.restore()
                db.session.commit()
                return {'message': 'User restored'}, 200
            return {'message': 'User is not deleted'}, 400
        return {'message': 'Invalid action'}, 400

        assert not sample_user.is_deleted
        
        sample_user.soft_delete()
        db.session.commit()

        """Test that restore clears the deleted_at timestamp."""
        sample_user.soft_delete()
        db.session.commit()

        db.session.commit()
